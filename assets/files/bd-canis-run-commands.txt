# === 1. Generate the PyRate input file (option 1: PBDB) - tutorial-1.md sections "Generate PyRate input file (option 1)" steps 3-6
# tutorial-1.md:31 - load the R utility script
source(".../PyRate-master/pyrate_utilities.r")
# tutorial-1.md:34 - declare the extant species (all others are taken as extinct)
extant_dogs = c("Canis rufus","Canis lupus","Canis aureus","Canis latrans","Canis mesomelas","Canis anthus","Pseudalopex gymnocercus","Canis adustus","Canis familiaris")
# tutorial-1.md:37 - parse the raw PBDB table and write *_PyRate.py + *_SpeciesList.txt
extract.ages.pbdb(file= ".../Canis_pbdb_data.csv",extant_species=extant_dogs)

# === 2. Check the generated input file
# tutorial-1.md:43 - browse to the PyRate folder
cd ".../PyRate-master"
# tutorial-1.md:45 - summary statistics of the data set
python PyRate.py '.../Canis_pbdb_data_PyRate.py' -data_info

# === 3. Generate the input file (option 2: hand-made occurrence table) and age replicates
# tutorial-1.md:68 - note the tutorial prints this one for Ursidae, not Canis; the Canis narrative later states its input file "included 10 randomized replicates" (tutorial-1.md:178)
extract.ages(file="…/PyRate/example_files/Ursidae.txt", replicates=10)

# === 4. Optional: check species names for typos / inconsistent spelling
# tutorial-1.md:98 - takes the *_SpeciesList.txt-style one-name-per-row text file
python PyRate.py  -check_names .../PBDB_dataset_TaxonList.txt

# === 5. Preservation model variants (section "Defining the preservation model")
# tutorial-1.md:126 - homogeneous Poisson process instead of the default NHPP
python PyRate.py .../Canis_pbdb_data_PyRate.py -mHPP
# tutorial-1.md:132 - time-variable Poisson process (TPP) with epoch boundaries
python PyRate.py .../Canis_pbdb_data_PyRate.py -qShift .../epochs_q.txt
# tutorial-1.md:137 - TPP with the prior rate parameter estimated from the data
python PyRate.py .../Canis_pbdb_data_PyRate.py -qShift .../epochs_q.txt -pP 1.5 0
# tutorial-1.md:144 - Gamma rate heterogeneity on the default NHPP
python PyRate.py .../Canis_pbdb_data_PyRate.py -mG
# tutorial-1.md:145 - Gamma heterogeneity under HPP
python PyRate.py .../Canis_pbdb_data_PyRate.py -mHPP -mG
# tutorial-1.md:146 - Gamma heterogeneity under TPP
python PyRate.py .../Canis_pbdb_data_PyRate.py -qShift .../epochs_q.txt -mG
# tutorial-1.md:151 - also log per-lineage relative preservation rates (writes ONE extra file)
python PyRate.py .../Canis_pbdb_data_PyRate.py -qShift .../epochs_q.txt -mG -log_sp_q_rates
# tutorial-1.md:156 - model test across NHPP / HPP / TPP (Gamma is NOT tested)
python PyRate.py .../Canis_pbdb_data_PyRate.py -qShift .../epochs_q.txt -PPmodeltest

# === 6. The BDMCMC birth-death analysis proper (section "Analysis setup")
# tutorial-1.md:172 - the algorithm is set explicitly to BDMCMC here
python PyRate.py .../Canis_pbdb_data_PyRate.py -A 2
# tutorial-1.md:180 - BDMCMC, HPP + Gamma preservation, first of the 10 age replicates
python PyRate.py .../Canis_pbdb_data_PyRate.py -A 2 -mHPP -mG -j 1
# tutorial-1.md:186 - longer run, sparser sampling: 20,000,000 iterations every 5,000 = 4,000 samples (tutorial-1.md:184-188)
python PyRate.py .../Canis_pbdb_data_PyRate.py -mG -n 20000000 -s 5000

# === 7. Summarize the run (section "Summarize the results") - details owned by ../pyrate-summarize-results
